-- ── MÓDULO 1: PREVISÃO DO TEMPO ──────────────────────────────

CREATE TABLE WTH_LOCATIONS (
    ID_LOCATION  NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    NM_CITY      VARCHAR2(100)  NOT NULL,
    NM_COUNTRY   VARCHAR2(50)   NOT NULL,
    NR_LATITUDE  NUMBER(10,6)   NOT NULL,
    NR_LONGITUDE NUMBER(10,6)   NOT NULL,
    NM_TIMEZONE  VARCHAR2(50)   DEFAULT 'America/Sao_Paulo',
    DT_CREATED   DATE           DEFAULT SYSDATE,
    CONSTRAINT UQ_LOCATION UNIQUE (NR_LATITUDE, NR_LONGITUDE)
);

CREATE TABLE WTH_HOURLY_FORECAST (
    ID_HOURLY         NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    ID_LOCATION       NUMBER        NOT NULL,
    DT_FORECAST       TIMESTAMP     NOT NULL,
    DT_COLLECTED      TIMESTAMP     DEFAULT SYSTIMESTAMP,
    NR_TEMPERATURE_2M NUMBER(5,2),
    NR_APPARENT_TEMP  NUMBER(5,2),
    NR_HUMIDITY       NUMBER(5,2),
    NR_DEWPOINT       NUMBER(5,2),
    NR_PRECIPITATION  NUMBER(6,2),
    NR_RAIN           NUMBER(6,2),
    NR_SNOWFALL       NUMBER(6,2),
    NR_PRECIP_PROB    NUMBER(5,2),
    NR_WEATHERCODE    NUMBER(4),
    NR_PRESSURE_MSL   NUMBER(7,2),
    NR_SURFACE_PRESS  NUMBER(7,2),
    NR_CLOUDCOVER     NUMBER(5,2),
    NR_VISIBILITY     NUMBER(10,2),
    NR_WINDSPEED_10M  NUMBER(6,2),
    NR_WINDDIR_10M    NUMBER(5,2),
    NR_WINDGUSTS      NUMBER(6,2),
    NR_UV_INDEX       NUMBER(4,2),
    NR_SHORTWAVE_RAD  NUMBER(8,2),
    CONSTRAINT FK_HOURLY_LOC FOREIGN KEY (ID_LOCATION) REFERENCES WTH_LOCATIONS(ID_LOCATION),
    CONSTRAINT UQ_HOURLY     UNIQUE (ID_LOCATION, DT_FORECAST)
);

CREATE TABLE WTH_DAILY_FORECAST (
    ID_DAILY           NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    ID_LOCATION        NUMBER        NOT NULL,
    DT_FORECAST        DATE          NOT NULL,
    DT_COLLECTED       TIMESTAMP     DEFAULT SYSTIMESTAMP,
    NR_TEMP_MAX        NUMBER(5,2),
    NR_TEMP_MIN        NUMBER(5,2),
    NR_APPARENT_MAX    NUMBER(5,2),
    NR_APPARENT_MIN    NUMBER(5,2),
    NR_SUNRISE         TIMESTAMP,
    NR_SUNSET          TIMESTAMP,
    NR_PRECIP_SUM      NUMBER(6,2),
    NR_RAIN_SUM        NUMBER(6,2),
    NR_PRECIP_HOURS    NUMBER(5,2),
    NR_PRECIP_PROB_MAX NUMBER(5,2),
    NR_WINDSPEED_MAX   NUMBER(6,2),
    NR_WINDGUSTS_MAX   NUMBER(6,2),
    NR_WINDDIR_DOM     NUMBER(5,2),
    NR_SHORTWAVE_SUM   NUMBER(10,2),
    NR_UV_MAX          NUMBER(4,2),
    NR_WEATHERCODE     NUMBER(4),
    CONSTRAINT FK_DAILY_LOC FOREIGN KEY (ID_LOCATION) REFERENCES WTH_LOCATIONS(ID_LOCATION),
    CONSTRAINT UQ_DAILY     UNIQUE (ID_LOCATION, DT_FORECAST)
);

CREATE TABLE WTH_COLLECTION_LOG (
    ID_LOG        NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    ID_LOCATION   NUMBER,
    DT_COLLECTION TIMESTAMP   DEFAULT SYSTIMESTAMP,
    NM_STATUS     VARCHAR2(10) CHECK (NM_STATUS IN ('SUCCESS','ERROR')),
    NR_RECORDS    NUMBER,
    DS_MESSAGE    VARCHAR2(500),
    CONSTRAINT FK_LOG_LOC FOREIGN KEY (ID_LOCATION) REFERENCES WTH_LOCATIONS(ID_LOCATION)
);

CREATE INDEX IDX_HOURLY_LOC_DT ON WTH_HOURLY_FORECAST (ID_LOCATION, DT_FORECAST DESC);
CREATE INDEX IDX_DAILY_LOC_DT  ON WTH_DAILY_FORECAST  (ID_LOCATION, DT_FORECAST DESC);
CREATE INDEX IDX_LOG_DT        ON WTH_COLLECTION_LOG  (DT_COLLECTION DESC);
